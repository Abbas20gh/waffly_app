    51→function PurchasesTab() {

    55→  const suppliers = useTable<Supplier>('suppliers')

    65→  const ppb = good?.piecesPerBox || 0
    66→  const boxMode = form.itemKind === 'GOOD' && form.unit === 'BOX' && ppb > 0
    67→  const qtyNum = parseFloat(form.quantity || '0')
    68→  const costNum = boxMode ? qtyNum * (parseFloat(form.pricePerBox || '0') || 0) : parseFloat(form.cost || '0')
    69→  const piecesNum = boxMode ? qtyNum * ppb : qtyNum
    70→
    71→  const switchKind = (kind: 'MATERIAL' | 'GOOD') => setForm(f => ({ ...f, itemKind: kind, materialId: '', unit: 'PIECE', quantity: '', cost: '', pricePerBox: '' }))
    72→
    73→  const save = async () => {
    74→    if (!form.materialId || qtyNum <= 0 || costNum <= 0) { toast({ title: 'قلم، مقدار و هزینه را وارد کنید', variant: 'destructive' }); return }
    75→    if (form.itemKind === 'GOOD' && form.unit === 'BOX' && ppb <= 0) { toast({ title: 'اول «تعداد در جعبه» را در تب کالاها تنظیم کنید', variant: 'destructive' }); return }

    79→      date: form.date,

    81→      quantity: piecesNum,
    82→      cost: costNum,
    83→      supplierId: form.supplierId || null,

    85→      paidAmount: Math.min(paid, costNum),
    86→      itemKind: form.itemKind,
    87→      boxesCount: boxMode ? qtyNum : 0,
    88→      note: form.note || null,
    89→      createdBy: getActiveUser() || null,
    90→      updatedAt: 0, deleted: 0,
    91→    })
    92→    toast({ title: 'خرید ثبت شد', description: `${(form.itemKind === 'GOOD' ? good?.name : mat?.name) || ''}: ${faDigits(piecesNum)} عدد — ${faMoney(costNum)} تومان` })
    93→    setOpen(false)
    94→    setForm(f => ({ ...f, quantity: '', cost: '', pricePerBox: '', paidAmount: '', note: '' }))
    95→  }
    96→
    97→  const list = [...purchases].filter(p => !p.deleted).sort((a, b) => (b.date + b.updatedAt).localeCompare(a.date + a.updatedAt)).slice(0, 50)

   100→  const supOf = (id: string | null) => suppliers.find(s => s.id === id)

   112→          ) : (
   113→            <div className="space-y-2">
   114→              {list.map(p => {
   115→                const isGood = p.itemKind === 'GOOD'
   116→                const item = isGood ? gOf(p.materialId) : matOf(p.materialId)
   117→                const st = effectiveSettled(p)
   118→                return (

   372→function GoodsTab() {
   373→  const goods = useTable<Good>('goods')
   374→  const [form, setForm] = useState({ name: '', piecesPerBox: '', minStock: '' })

   458→function SuppliersTab() {
   459→  const suppliers = useTable<Supplier>('suppliers')
   460→  const purchases = useTable<Purchase>('purchases')
   461→  const [form, setForm] = useState({ name: '', phone: '', address: '' })
   462→
   463→  const save = async () => {
   464→    if (!form.name.trim()) { toast({ title: 'نام تامین‌کننده لازم است', variant: 'destructive' }); return }
   465→    await putRecord<Supplier>('suppliers', {
   466→      id: uid(), name: form.name.trim(), phone: form.phone || null, address: form.address || null,
   467→      updatedAt: 0, deleted: 0,
   468→    })
   469→    setForm({ name: '', phone: '', address: '' })
   470→    toast({ title: 'تامین‌کننده ذخیره شد' })
   471→  }
   472→
   473→  const stats = useMemo(() => {
   474→    const m = new Map<string, { count: number; total: number; due: number }>()
   475→    for (const p of active(purchases)) {
   476→      if (!p.supplierId) continue
   477→      const st = m.get(p.supplierId) || { count: 0, total: 0, due: 0 }
   478→      st.count++
   479→      st.total += p.cost || 0
   480→      st.due += Math.max(0, (p.cost || 0) - (p.paidAmount || 0))
   481→      m.set(p.supplierId, st)
   482→    }
   483→    return m
   484→  }, [purchases])
   485→
   486→  return (
   487→    <div className="grid lg:grid-cols-5 gap-4">
   488→      <Card className="waffly-card lg:col-span-2 h-fit">
   489→        <CardHeader className="pb-2"><CardTitle className="text-sm flex items-center gap-2"><Truck className="h-4 w-4" /> تامین‌کننده جدید</CardTitle></CardHeader>
   490→        <CardContent className="space-y-3">
   491→          <FormRow label="نام"><Input value={form.name} onChange={e => setForm(f => ({ ...f, name: e.target.value }))} className="h-11" /></FormRow>
   492→          <FormRow label="تلفن"><Input inputMode="tel" value={form.phone} onChange={e => setForm(f => ({ ...f, phone: e.target.value }))} className="waffly-num h-11" dir="ltr" /></FormRow>
   493→          <FormRow label="آدرس"><Input value={form.address} onChange={e => setForm(f => ({ ...f, address: e.target.value }))} className="h-11" /></FormRow>
   494→          <Button className="w-full h-11" onClick={save}>ذخیره</Button>
   495→        </CardContent>
   496→      </Card>
   497→      <Card className="waffly-card lg:col-span-3">
   498→        <CardHeader className="pb-2"><CardTitle className="text-sm">تامین‌کنندگان ({faDigits(active(suppliers).length)})</CardTitle></CardHeader>
   499→        <CardContent>
   500→          {active(suppliers).length === 0 ? (
   501→            <EmptyState title="تامین‌کننده‌ای ثبت نشده" icon={<Truck className="h-5 w-5" />} />
   502→          ) : (
   503→            <div className="space-y-1.5">
   504→              {active(suppliers).map(s => {
   505→                const st = stats.get(s.id)
   506→                return (
   507→                  <div key={s.id} className="flex items-center gap-3 rounded-lg border px-3 py-2.5">
   508→                    <div className="flex-1 min-w-0">
   509→                      <p className="text-sm font-semibold">{s.name}</p>
   510→                      <p className="text-[11px] text-muted-foreground waffly-num">
   511→                        {s.phone && <span dir="ltr">{s.phone}</span>}
   512→                        {st && ` • ${faDigits(st.count)} خرید`}
   513→                      </p>
   514→                    </div>
   515→                    {st && (
   516→                      <div className="text-left shrink-0">
   517→                        {st.due > 0.5 && <p className="text-[11px] text-red-600 font-bold waffly-num">بدهی: {faMoney(st.due)}</p>}
   518→                        <p className="text-[10px] text-muted-foreground waffly-num">مجموع: {faMoney(st.total)}</p>
   519→                      </div>
   520→                    )}
   521→                    <Button variant="ghost" size="icon" className="h-8 w-8 text-muted-foreground hover:text-red-600" aria-label="حذف"
   522→                      onClick={() => void removeRecord('suppliers', s.id)}>
   523→                      <Trash2 className="h-4 w-4" />
   524→                    </Button>
   525→                  </div>
   526→                )
   527→              })}
   528→            </div>
   529→          )}
   530→        </CardContent>
   531→      </Card>
   532→    </div>
   533→  )
