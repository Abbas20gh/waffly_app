    57→    57→function SalesTab() {
    58→    58→  const sales = useTable<Sale>('sales')
    59→    59→  const customers = useTable<Customer>('customers')
    60→    60→  const breadTypes = useTable<BreadType>('breadTypes')
    61→    61→  const [open, setOpen] = useState(false)
    62→    62→  const [filter, setFilter] = useState<'all' | 'unpaid' | 'checks'>('all')
    63→    63→
    64→    64→  const emptyItem: SaleItem = { breadTypeId: '', qty: 0, unitPrice: 0, delivered: 0, returned: 0, returnCost: 0 }
    65→    65→  const [form, setForm] = useState({
    66→    66→    date: todayJalali(),
    67→    67→    customerId: '',
    68→    68→    items: [{ ...emptyItem } as SaleItem],
    69→    69→    settledStatus: 'UNPAID' as Sale['settledStatus'],
    70→    70→    paidAmount: '',
    71→    71→    paymentMethod: 'CASH' as Sale['paymentMethod'],
    72→    72→    checkDueDate: '',
    73→    73→    checkNumber: '',
    74→    74→    checkBank: '',
    75→    75→    paymentDate: '',
    76→    76→    note: '',
    77→    77→  })
    78→    78→
    79→    79→  const itemsTotal = form.items.reduce((a, it) => a + (it.qty || 0) * (it.unitPrice || 0), 0)
    80→    80→  const returnTotal = form.items.reduce((a, it) => a + (it.returnCost || 0), 0)
    81→    81→  const grandTotal = Math.max(0, itemsTotal - returnTotal)
    82→    82→
    83→    83→  const openNew = () => {
    84→    84→    setForm({
    85→    85→      date: todayJalali(),
    86→    86→      customerId: '',
    87→    87→      items: [{ ...emptyItem }],
    88→    88→      settledStatus: 'UNPAID',
    89→    89→      paidAmount: '',
    90→    90→      paymentMethod: 'CASH',
    91→    91→      checkDueDate: '',
    92→    92→      checkNumber: '',
    93→    93→      checkBank: '',
    94→    94→      paymentDate: '',
    95→    95→      note: '',
    96→    96→    })
    97→    97→    setOpen(true)
    98→    98→  }
    99→    99→
   100→   100→  const updateItem = (idx: number, patch: Partial<SaleItem>) => {
   101→   101→    setForm(f => {
   102→   102→      const items = f.items.map((it, i) => i === idx ? { ...it, ...patch } : it)
   103→   103→      return { ...f, items }
   104→   104→    })
   105→   105→  }
   106→   106→
   107→   107→  const save = async () => {
   108→   108→    if (!form.customerId) { toast({ title: 'مشتری را انتخاب کنید', variant: 'destructive' }); return }
   109→   109→    const items = form.items.filter(it => it.breadTypeId && (it.qty > 0 || it.delivered > 0))
   110→   110→    if (items.length === 0) { toast({ title: 'حداقل یک قلم کالا اضافه کنید', variant: 'destructive' }); return }
   111→   111→    const paid = form.settledStatus === 'PAID' ? grandTotal : parseFloat(form.paidAmount || '0')
   112→   112→    const status: Sale['settledStatus'] = form.settledStatus === 'PAID' || paid >= grandTotal - 0.5
   113→   113→      ? 'PAID'
   114→   114→      : paid > 0.5 ? 'PARTIAL' : 'UNPAID'
   115→   115→    await putRecord<Sale>('sales', {
   116→   116→      id: uid(),
   117→   117→      date: form.date,
   118→   118→      customerId: form.customerId,
   119→   119→      items: JSON.stringify(items),
   120→   120→      totalAmount: grandTotal,
   121→   121→      settledStatus: status,
   122→   122→      paidAmount: Math.min(paid, grandTotal),
   123→   123→      paymentMethod: form.paymentMethod,
   124→   124→      checkDueDate: form.paymentMethod === 'CHECK' ? (form.checkDueDate || null) : null,
   125→   125→      checkNumber: form.paymentMethod === 'CHECK' ? (form.checkNumber || null) : null,
   126→   126→      checkBank: form.paymentMethod === 'CHECK' ? (form.checkBank || null) : null,
   127→   127→      paymentDate: status === 'PAID' ? (form.paymentDate || null) : (form.paymentDate || null),
   128→   128→      note: form.note || null,
   129→   129→      createdBy: getActiveUser() || null,
   130→   130→      updatedAt: 0,
   131→   131→      deleted: 0,
   132→   132→    })
   133→   133→    toast({ title: 'فروش ثبت شد', description: `${faMoney(grandTotal)} تومان` })
   134→   134→    setOpen(false)
   135→   135→  }
   136→   136→
   137→   137→  const quickSettle = async (s: Sale) => {
   138→   138→    await putRecord<Sale>('sales', { ...s, settledStatus: 'PAID', paidAmount: s.totalAmount, paymentDate: s.paymentDate || todayJalali() })
   139→   139→    toast({ title: 'تسویه شد', description: `${faMoney(saleDue(s))} تومان` })
   140→   140→  }
   141→   141→
   142→   142→  const list = [...sales]
   143→   143→    .filter(s => !s.deleted)
   144→   144→    .filter(s => filter === 'all' || (filter === 'unpaid' ? effectiveSettled(s) !== 'PAID' : s.paymentMethod === 'CHECK'))
   145→   145→    .sort((a, b) => (b.date + b.updatedAt).localeCompare(a.date + a.updatedAt))
   146→   146→    .slice(0, 60)
   147→   147→  const cName = (id: string) => customers.find(c => c.id === id)?.name || 'نامشخص'
   148→   148→  const btLabel = (id: string) => {
   149→   149→    const b = breadTypes.find(x => x.id === id)
   150→   150→    return b ? b.name : 'کالا'
   151→   151→  }
   152→   152→
   153→   153→  return (
   154→   154→    <div className="space-y-4">
   155→   155→      <div className="flex flex-wrap items-center justify-between gap-2">
   156→   156→        <div className="flex gap-1.5">
   157→   157→          {([['all', 'همه'], ['unpaid', 'تسویه‌نشده'], ['checks', 'چکی']] as const).map(([k, label]) => (
   158→   158→            <button
   159→   159→              key={k}
   160→   160→              onClick={() => setFilter(k)}
   161→   161→              className={cn('rounded-lg px-3 py-1.5 text-xs font-medium border min-h-9',
   162→   162→                filter === k ? 'bg-primary text-primary-foreground border-primary' : 'bg-card text-muted-foreground')}
   163→   163→            >
   164→   164→              {label}
   165→   165→            </button>
   166→   166→          ))}
   167→   167→        </div>
   168→   168→        <Button onClick={openNew} className="h-11"><Plus className="ml-1 h-4 w-4" /> فروش جدید</Button>
   169→   169→      </div>
   170→   170→
   171→   171→      <Card className="waffly-card">
   172→   172→        <CardContent className="p-3">
   173→   173→          {list.length === 0 ? (
   174→   174→            <EmptyState title="فروشی یافت نشد" desc="با دکمه «فروش جدید» اولین فروش را ثبت کنید." icon={<ShoppingCart className="h-5 w-5" />} />
   175→   175→          ) : (
   176→   176→            <div className="space-y-2">
   177→   177→              {list.map(s => {
   178→   178→                const items = (() => { try { return JSON.parse(s.items || '[]') as SaleItem[] } catch { return [] } })()
   179→   179→                const st = effectiveSettled(s)
   180→   180→                return (
   181→   181→                  <div key={s.id} className="rounded-xl border p-3 space-y-2">
   182→   182→                    <div className="flex flex-wrap items-center gap-2">
   183→   183→                      <p className="text-sm font-bold">{cName(s.customerId)}</p>
   184→   184→                      <span className="text-[11px] text-muted-foreground waffly-num">{prettyJalali(s.date)}</span>
   185→   185→                      <span className="text-[10px] rounded bg-muted px-1.5 py-0.5">{PAY_METHODS.find(p => p.key === s.paymentMethod)?.label}</span>
   186→   186→                      {s.paymentMethod === 'CHECK' && s.checkDueDate && (
   187→   187→                        <span className="text-[10px] rounded bg-indigo-50 text-indigo-700 border border-indigo-200 px-1.5 py-0.5 waffly-num">چک: {prettyJalali(s.checkDueDate)}</span>
   188→   188→                      )}
   189→   189→                      <div className="flex-1" />
   190→   190→                      <SettleBadge status={st} paid={s.paidAmount} total={s.totalAmount} />
   191→   191→                      <Money value={s.totalAmount} className="font-bold text-sm" />
   192→   192→                      <span className="text-[10px] text-muted-foreground">تومان</span>
   193→   193→                    </div>
   194→   194→                    <p className="text-[11px] text-muted-foreground waffly-num">
   195→   195→                      {items.map((it, i) => `${btLabel(it.breadTypeId)} × ${faDigits(it.delivered || it.qty)}`).join(' • ')}
   196→   196→                      {s.paymentDate && ` • وصول: ${prettyJalali(s.paymentDate)}`}
   197→   197→                      {s.createdBy && ` • ${s.createdBy}`}
   198→   198→                    </p>
   199→   199→                    {st !== 'PAID' && (
   200→   200→                      <div className="flex items-center gap-2 pt-1">
   201→   201→                        <Button size="sm" variant="outline" className="h-8 text-[11px]" onClick={() => void quickSettle(s)}>
   202→   202→                          <Check className="h-3.5 w-3.5" /> تسویه کامل
   203→   203→                        </Button>
   204→   204→                        <span className="text-[11px] text-red-600 waffly-num">مانده: {faMoney(saleDue(s))}</span>
   205→   205→                      </div>
   206→   206→                    )}
   207→   207→                  </div>
   208→   208→                )
   209→   209→              })}
   210→   210→            </div>
   211→   211→          )}
   212→   212→        </CardContent>
   213→   213→      </Card>
   214→   214→
   215→   215→      <SaleFormDialog
   216→   216→        open={open} onOpenChange={setOpen}
   217→   217→        form={form} setForm={setForm}
   218→   218→        customers={active(customers)} breadTypes={active(breadTypes)}
   219→   219→        itemsTotal={itemsTotal} returnTotal={returnTotal} grandTotal={grandTotal}
   220→   220→        onSave={save}
   221→   221→        onQuickAddCustomer={async (name) => {
   222→   222→          const id = uid()
   223→   223→          await putRecord<Customer>('customers', { id, name, phone: null, address: null, cooperationType: null, updatedAt: 0, deleted: 0 })
   224→   224→          setForm(f => ({ ...f, customerId: id }))
   225→   225→          toast({ title: 'مشتری اضافه شد', description: name })
   226→   226→        }}
   227→   227→      />
   228→   228→    </div>
   229→   229→  )
   230→   230→}

   234→   234→  onOpenChange: (v: boolean) => void

   241→   241→  setForm: React.Dispatch<React.SetStateAction<{

   285→   285→              className="h-10"
   286→   286→            />
   287→   287→            <Button
   288→   288→              type="button" variant="outline" className="h-10 shrink-0"
   289→   289→              disabled={!newCustomer.trim()}
   290→   290→              onClick={() => { onQuickAddCustomer(newCustomer.trim()); setNewCustomer('') }}
   291→   291→            >
   292→   292→              <Plus className="h-4 w-4" /> افزودن
   293→   293→            </Button>
   294→   294→          </div>

   328→   328→                  </label>

   496→   496→function ChecksTab() {

   501→   501→    .sort((a, b) => (a.checkDueDate || a.date).localeCompare(b.checkDueDate || b.date))
   502→   502→  const cName = (id: string) => customers.find(c => c.id === id)?.name || 'نامشخص'
   503→   503→

   505→   505→    await putRecord<Sale>('sales', { ...s, settledStatus: 'PAID', paidAmount: s.totalAmount, paymentDate: todayJalali() })
   506→   506→    toast({ title: 'چک وصول شد' })
   507→   507→  }
   508→   508→
   509→   509→  const statusChip = (s: Sale) => {
   510→   510→    const st = effectiveSettled(s)
   511→   511→    if (st === 'PAID') return <span className="rounded bg-green-50 text-green-700 border border-green-200 px-2 py-0.5 text-[10px]">وصول‌شده</span>
   512→   512→    const days = daysSince(s.checkDueDate || s.date)
   513→   513→    if (s.checkDueDate && days > 0 || (!s.checkDueDate && days > 0)) {
   514→   514→      return <span className="rounded bg-red-50 text-red-700 border border-red-200 px-2 py-0.5 text-[10px] waffly-num">{faDigits(days)} روز گذشته</span>
   515→   515→    }

   517→   517→    return <span className="rounded bg-muted px-2 py-0.5 text-[10px] waffly-num">{faDigits(-days)} روز مانده</span>
   518→   518→  }
   519→   519→
   520→   520→  return (
   521→   521→    <Card className="waffly-card">
   522→   522→      <CardHeader className="pb-2">
   523→   523→        <CardTitle className="text-sm flex items-center gap-2"><Landmark className="h-4 w-4" /> فروش‌های چکی</CardTitle>

   534→   534→                  <p className="text-[11px] text-muted-foreground waffly-num">
